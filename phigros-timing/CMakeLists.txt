cmake_minimum_required(VERSION 3.18)
project(phigros_timing LANGUAGES C CXX ASM)
set(CMAKE_CXX_STANDARD 17)
add_library(phigrostiming SHARED timing.cpp)
target_include_directories(phigrostiming PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
target_link_libraries(phigrostiming PRIVATE log dl)
target_link_options(phigrostiming PRIVATE -Wl,-z,max-page-size=16384 -Wl,--exclude-libs,ALL)
add_executable(timing_selftest timing.cpp speed_clock_test.cpp)
target_compile_definitions(timing_selftest PRIVATE PHIGROS_TIMING_SELF_TEST)
target_include_directories(timing_selftest PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
target_link_libraries(timing_selftest PRIVATE log dl)
target_link_options(timing_selftest PRIVATE -Wl,-z,max-page-size=16384)
